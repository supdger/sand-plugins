$listener = Get-NetTCPConnection -State Listen -LocalPort {{frontendPort}}
$listener | Select-Object LocalAddress, OwningProcess
Get-Process -Id ($listener.OwningProcess | Sort-Object -Unique)
Stop-Process -Id ($listener.OwningProcess | Sort-Object -Unique) -Confirm
